\begin{abstract}
本文确定深区非边界局部化商 $Q_c=\mathcal T_{f_c}M_{a,b}$ 的局部幂零部分
$G_c=\{v\in Q_c:E^nv=0\ \text{对某个}\ n\}$（$E=e_{-c}$）的完整结构。
在谱流扭转后的基本图像（$E=e_0$，$F=f_0$，$H=h_0$，$A\le-1$，$B\ge L$，
$\mu\in\Z$）中，证明：$G_c$ 是有限维 $\slc(E,H,F)$-不可约模的直和；权为
$\nu$ 的 $E$-原初空间由顶层条件 $T^{\nu+1}(X)u=0$ 精确描述；$\g$-作用完全由
原初投影算子编码。结论：$G_c$ 不单、$\operatorname{soc}(G_c)=0$（从而
$\operatorname{soc}(Q_c)=0$）、合成长度无限；其标准权滤过的商为单模
$P_\nu^{(q)}\otimes L(\nu)$ 加上权零处的一个平凡商。浅区中 $G_c=0$。
\end{abstract}

\section{背景与 $\slc$-结构}
沿用统一笔记与 addendum 的记号。深区（$c<b$，$c\le-a-1$，即
$d_*=b-c\ge L$）中，谱流 $\sigma_{-c}$ 将 $Q_c$ 等同于
$Q=\mathcal T_{f_0}M_{A,B}(\chi)$，其中 $A=a+c\le-1$，$B=b-c=d_*\ge L$，
$\chi(h_0)=\mu=\lambda_0-c\kappa\in\Z$。记 $E=e_0$，$F=f_0$，$H=h_0$，
$G=\{v\in Q:E^nv=0\ \text{对某个}\ n\ge1\}$。自由 PBW 部分为
\[
\mathcal X=\Big\{f_{j_1}\cdots f_{j_p}\,e_{i_1}\cdots e_{i_q}\,
h_{r_1}\cdots h_{r_s}:\ j_t\le B,\ j_t\ne0,\ i_t\le A,\ r_u<0\Big\},
\]
$Q$ 的基为 $\{F^{-m}Xu: m\ge1,\ X\in\mathcal X\}$。记 $\deg X=2q-2p$，则
$H(F^{-m}Xu)=(\mu+2m+\deg X)F^{-m}Xu$；权空间 $Q_\nu$，$\nu\in\mu+2\Z$。

\begin{theorem}\label{thm:sl2}
$G$ 的每个向量都落在有限维 $\slc(E,H,F)$-子模中，且
\[
G\;\cong\;\bigoplus_{\nu\in\Z_{\ge0}} P_\nu\otimes L(\nu),
\qquad P_\nu:=\ker E\cap G\cap Q_\nu,
\]
其中 $L(\nu)$ 是 $(\nu+1)$ 维不可约 $\slc$-模。特别地，$G$ 的
$E$-原初向量之权属于 $\Z_{\ge0}$，$P_\nu$ 是 $L(\nu)$ 的重数空间。
\end{theorem}
\begin{proof}
$E$ 在 $G$ 上局部幂零（定义），$F$ 在 $Q$ 上局部幂零，故任一向量生成的
$\slc$-张成由有限个 $F^iE^jv$ 张成，有限维；完全可约是 Weyl 定理。原初
向量 $p$ 满足 $F^Np=0$ 时生成 $L(\nu)$，$\nu$ 为 $p$ 的 $H$-权，
$F^Np=0$ 迫使 $\nu\le N-1$，故 $\nu\in\Z_{\ge0}$。
\end{proof}

\section{E-作用与原初空间}
\begin{lemma}\label{lem:Eaction}
在 $Q$ 的 PBW 基上，
\[
E\bigl(F^{-m}Xu\bigr)
=F^{-m}\,T(X)u
-m\bigl(\mu+\deg X+m+1\bigr)F^{-m-1}Xu,
\qquad
T:=\pi_0\circ\ad(e_0),
\]
其中 $\pi_0$ 是 $M$ 到其 $F^0$-部分的投影。$T$ 把 $\mathcal X$-字映为
$\mathcal X$-字的组合，使 $\deg$ 增加 $2$，并保持消灭 $u$ 的字的理想：
$Wu=0\Rightarrow T(W)u=0$。
\end{lemma}
\begin{proof}
由三重恒等式 $EF^{-m}=F^{-m}E-mF^{-m-1}H-m(m+1)F^{-m-2}F$ 与 $Eu=0$ 得
$E(F^{-m}Xu)=F^{-m}(\ad E)(X)u-m(\mu+\deg X)F^{-m-1}Xu
-m(m+1)F^{-m-2}FXu$。在局部化中 $F^{-m-2}FXu=F^{-m-1}Xu$，且
$(\ad E)(X)u\in M$ 在 $Q$ 中只贡献 $F^0$-部分，公式成立。括号关系
$[e_0,f_j]=h_j$、$[e_0,h_r]=-2e_r$、$[e_0,e_i]=0$ 给出 $T$ 的字级描述；
最后 $T(W)u=\pi_0E(Wu)$。
\end{proof}

\begin{theorem}\label{thm:P}
对 $\nu\in\Z_{\ge0}$，
\[
P_\nu\;\cong\;
\Big\{X\in\mathcal X:\ \deg X=-(\mu+\nu+2),\quad
T^{\nu+1}(X)u=0\Big\},
\]
同构为 $v\mapsto$ 其在 pole $\nu+1$ 的顶层。特别地 $P_\nu\ne0$ 迫使
$\mu+\nu$ 为偶数，且每个非零 $P_\nu$ 无穷维（乘 $h_{-1}$ 的幂）。
\end{theorem}
\begin{proof}
写 $v=\sum_{m=1}^{N}F^{-m}X_mu\in Q_\nu$，则 $\deg X_m=\nu-\mu-2m$。由引理
\ref{lem:Eaction}，
\[
Ev=\sum_{m}F^{-m}T(X_m)u-\sum_{m}m(\nu-m+1)F^{-m-1}X_mu.
\]
在 pole $N+1$ 只有对角项，故 $Ev=0$ 迫使 $N=\nu+1$ 且 $X_{\nu+1}$ 无约束；
pole $m=1,\dots,\nu+1$ 的方程为
\[
T(X_m)u=(m-1)(\nu-m+2)X_{m-1}u,\qquad X_0=0,\ X_{\nu+2}=0.
\]
对 $m\ge2$ 系数 $(m-1)(\nu-m+2)\ne0$，故各层由顶层唯一决定，而剩余条件
$T(X_1)u=0$ 恰为 $T^{\nu+1}(X_{\nu+1})u=0$。
\end{proof}

\begin{remark}\label{rem:checks}
$\mu=-2$ 时 $P_0=\C w_1$（$X=1$），与 $Ew_1=0$ 一致；$e_{-1}w_1\in P_2$ 的
顶层是 pole $3$ 处的 $-2f_{-1}$，且 $T^3(f_{-1})=0$。基本模型 $\lambda_0=0$
时 $e_0^2f_1^2w_1\in P_2$，顶层为 $f_1^2$，$T^3(f_1^2)u=(-12h_1e_1+4e_2)u=0$。
\end{remark}

\section{原初投影}
设 $p\in P_\nu$ 的顶层为 $X$，各层为 $X_m$。
\begin{proposition}\label{prop:proj}
\begin{enumerate}[label=(\roman*)]
\item $i\le A$：$e_ip\in P_{\nu+2}$，顶层为 $e_iX$；记 $\hat e_i(X)=e_iX$。
\item $r<0$、$\nu\ge1$：
$h_rp+\tfrac{2}{\nu+2}F e_r p\in P_\nu$，顶层为
\[
\hat h_r(X)=\frac{\nu}{\nu+2}\Bigl(-h_rX+\tfrac{1}{\nu}f_rT(X)\Bigr).
\]
$\nu=0$ 时 $h_rp=-Fe_rp$ 落在 $e_rp\in P_2$ 的 $L(2)$-块中，故 $h_rp$ 的
$P_0$-分量为零。
\item $j\ge1$：$f_jp\in G$，$E^2f_jp=-2e_jp$，其原初投影落在 $P_{\nu+2}$，
顶层为 $e_jX$（低层由递推决定）。
\end{enumerate}
$G$ 作为 $\g$-模由 $P=\bigoplus_\nu P_\nu$ 生成；任一 $0\ne p\in P_\nu$
生成的子模包含 $L(\nu)p$ 及 $p$ 在 $\{\hat e_i,\hat h_r,\hat f_j,E,F\}$
所有字下的像。
\end{proposition}
\begin{proof}
(i) $[E,e_i]=0$，对 $e_ip$ 应用定理 \ref{thm:P} 的 pole 分析即得。
(ii) $[E,h_r]=-2e_r$、$[f_0,e_r]=-h_r$（$r<0$）给出
$h_rp=e_rFp-Fe_rp$；$\nu=0$ 时 $Fp=0$，故 $h_rp=-Fe_rp$；$\nu\ge1$ 时
$EFe_rp=(\nu+2)e_rp$，故 $E(h_rp+\tfrac{2}{\nu+2}Fe_rp)=0$，pole $\nu+2$
层恰由 $[e_r,F^{-m}]=-mF^{-m-1}h_r-m(m+1)F^{-m-2}f_r$ 消去。
(iii) $[E,f_j]=h_j$ 给出 $E^2f_jp=-2e_jp$，再由 $[E,e_j]=0$ 得
$e_jp\in P_{\nu+2}$。
\end{proof}

\section{权滤过与子模结构}
定理 \ref{thm:P} 把 $G$ 的子模问题转化为权记簿问题。对 $\nu\ge0$ 定义
\[
G^{\ge\nu}\;=\;
\bigoplus_{\nu'\ge\nu} P_{\nu'}\otimes L(\nu')\;\subseteq\;G ,
\]
即顶层权至少为 $\nu$ 的 $\slc$-块的张成。

\begin{lemma}\label{lem:stable}
每个 $G^{\ge\nu}$ 是 $G$ 的 $\g$-子模。$P_\nu\ne0$ 当且仅当
$\nu\ge\nu_0$ 且 $\nu\equiv\mu\pmod2$（$\nu_0$ 为允许顶层的极小 $\nu$），
且对每个这样的 $\nu$，$e_1:P_\nu\to P_{\nu+2}$ 非零。
\end{lemma}
\begin{proof}
生成元对块的作用：$e_r$ 把 $P_{\nu'}$ 映到 $P_{\nu'+2}$；$h_r$、$f_j$ 有
$P_{\nu'}$-分量（命题 \ref{prop:proj} 的投影）与 $P_{\nu'+2}$-分量；
$E,H,F$ 在块内作用，$K$ 为数乘。故 $G^{\ge\nu}$ 稳定。顶层之权迫使
$\mu+\nu$ 为偶数；对 $0\ne p$，$e_1p$ 的顶层 $f_1X\ne0$，故支撑无空缺。
\end{proof}

\begin{theorem}\label{thm:weight}
以 $\nu_0$ 如引理 \ref{lem:stable}，
\[
G=G^{\ge\nu_0}\;\supsetneq\;G^{\ge\nu_0+1}\;\supsetneq\;
G^{\ge\nu_0+2}\;\supsetneq\;\cdots
\]
是 $\g$-子模的无限严格降链（$P_\nu=0$ 的层合并），其商为：
\begin{enumerate}[label=(\roman*)]
\item $\nu\ge1$ 时
\[
G^{\ge\nu}/G^{\ge\nu+1}
\;\cong\;\bigoplus_{q\ge q_\nu}
\bigl(P_\nu^{(q)}\otimes L(\nu)\bigr),
\qquad
P_\nu^{(q)}=\bigl\{X\in\mathcal X:\ p(X)=q+\tfrac{\mu+\nu+2}{2},\ 
e(X)=q,\ T^{\nu+1}(X)u=0\bigr\},
\]
且每个直和项都是单 $\g$-模。
\item 若 $P_0\ne0$，则 $G^{\ge0}/G^{\ge1}\cong P_0\otimes L(0)$ 是一维平凡
$\g$-模的直和：对 $p\in P_0$ 有 $h_rp=-Fe_rp$、$h_jp=-Fe_jp$、
$f_jp=-\tfrac12F^2e_jp$，故每个生成元都把 $P_0$ 映到 $\nu\ge2$ 部分，且
$Fp=Ep=0$。
\end{enumerate}
因此：
\begin{enumerate}[label=(\roman*)]
\item $G$ 不单；
\item $\operatorname{soc}(G)=0$，从而 $\operatorname{soc}(Q_c)=0$：$G_c$ 不是
$Q_c$ 的 socle；
\item $G$ 没有有限合成列（长度无限），合成因子为
$P_\nu^{(q)}\otimes L(\nu)$ 与 $P_0\otimes L(0)$；
\item $G$ 的头是 $\nu_0$-商：单模的直和（$\nu_0=0$ 时为平凡模的直和）。
\end{enumerate}
\end{theorem}
\begin{proof}
链的严格性与非空性来自引理 \ref{lem:stable}。商
$G^{\ge\nu}/G^{\ge\nu+1}$ 中，$e_r$ 把直和项的顶向量送入 $G^{\ge\nu+2}$，
并在低块层上经投影 $\hat h_r,\hat f_r$ 作用（由
$[e_r,F^s]=sh_rF^{s-1}-s(s-1)f_rF^{s-2}$），故保持每个直和项。投影
$\hat h_r,\hat h_j,\hat f_j$ 保持顶层的 $(p,e)$ 计数（其候选低计项因
$\deg X=-(\mu+\nu+2)$ 而在 $P_\nu$ 内相消）。h-投影在
$(h$-次数、e-次数$)$ 滤过下三角作用，首项 $h_rX$ 的非零系数为
$-\nu/(\nu+2)$（$\nu\ge1$ 时成立），故由任一非零类可生成整个直和项
$P_\nu^{(q)}\otimes L(\nu)$；$L(\nu)$-因子在 $E,H,F$ 下不可约。故每个直和项
单，不同 $q$ 之间无交互。$\nu=0$ 时 (ii) 中的恒等式（基本模型情形已由
(ii) 中的恒等式对一般情形成立：对 $p_0=F^{-1}Xu\in P_0$ 有
$h_rp_0=F^{-1}h_rXu+2F^{-2}f_rXu=-Fe_rp_0$，且
$f_jp_0+\tfrac12F^2e_jp_0=-\tfrac12h_jXu=
-\tfrac12(\lambda_jX+[h_j,X])u\in M$（$[h_j,X]u$ 被吸收进 $M$）；基本模型
情形已由 \texttt{work/check\_gc\_filter.py} 验证。
含原初向量，从而（由生成论证）含整个尾部 $G^{\ge\nu}$，故不存在单子模，
socle 为零。
\end{proof}

\begin{remark}\label{rem:shallow}
\begin{enumerate}[label=(\roman*)]
\item 浅区 $-a\le c<b$ 中 $E=e_{-c}$ 在 $Q_c$ 上单射，故
$G_c=\{v:E^nv=0\}=0$：以上结构论是对所有非边界指标的完整回答，只有深区
带有非零局部幂零部分。
\item 基本模型 $M_{-1,1}$ 在 $\lambda_0=0$ 时 $P_0=0$、$\nu_0=2$：链为
$G\supsetneq G^{\ge4}\supsetneq G^{\ge6}\supsetneq\cdots$，典范向量
$e_0^2f_1^2w_1$（顶层 $f_1^2$）生成第一个商的和项 $P_2^{(0)}\otimes L(2)$。
\item $\mu=-2$ 且 $B=L$、各 $\lambda_j\ne0$（$1\le j\le L$）时
$P_0=\C w_1$，链从 $\nu_0=0$ 开始，首个商为平凡模 $\C w_1$。
\end{enumerate}
\end{remark}
